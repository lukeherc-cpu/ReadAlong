cmake_minimum_required(VERSION 3.22.1)
project(readalong_whistle LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The pinned Whistle Android runtime supports arm64-v8a. Use an ARM64 device or an emulator with ARM translation.")
endif()
set(NEEDLE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/needle/prebuilt/android-arm64)
add_library(needle STATIC IMPORTED)
set_target_properties(needle PROPERTIES IMPORTED_LOCATION ${NEEDLE_DIR}/libneedle.a)
add_library(readalong_whistle SHARED whistle_jni.cpp)
target_include_directories(readalong_whistle PRIVATE ${NEEDLE_DIR})
target_compile_options(readalong_whistle PRIVATE -O3)
target_link_libraries(readalong_whistle PRIVATE needle android log)
target_link_options(readalong_whistle PRIVATE "-Wl,-z,max-page-size=16384")
# Answer Whistle's CPU-topology reads from the CPUs actually allowed (see whistle_jni.cpp).
target_link_options(readalong_whistle PRIVATE "-Wl,--wrap=fopen" "-Wl,--wrap=_ZNSt6__ndk16thread20hardware_concurrencyEv")
